import argparse
import json
import os
from agents import GenerationAgent, ReflectionAgent, RankingAgent, ProximityAgent, EvolutionAgent, MetaReviewAgent, Supervisor
from utils import load_experiment_file, save_ideas_to_json

def main():
    parser = argparse.ArgumentParser(description="AI Co-Scientist for Machine Learning Experiment Modification")
    parser.add_argument("--experiment", required=True, help="Name of the experiment (directory in templates/)")
    parser.add_argument("--num_ideas", type=int, default=5, help="Number of top ideas to save")
    parser.add_argument("--skip_lit_review", action="store_true", help="Skip literature review steps (for faster testing)")
    parser.add_argument("--iterations", type=int, default=3, help="Number of iterations for the co-scientist loop")
    args = parser.parse_args()

    experiment_path = os.path.join("templates", args.experiment)
    experiment_file = os.path.join(experiment_path, "experiment.py")
    ideas_file = os.path.join(experiment_path, "ideas.json")

    if not os.path.exists(experiment_file):
        print(f"Error: Experiment file not found: {experiment_file}")
        return

    experiment_code = load_experiment_file(experiment_file)

    # Initialize agents
    generation_agent = GenerationAgent(experiment_code, skip_lit_review=args.skip_lit_review)
    reflection_agent = ReflectionAgent(experiment_code, skip_lit_review=args.skip_lit_review)
    ranking_agent = RankingAgent()
    proximity_agent = ProximityAgent()
    evolution_agent = EvolutionAgent(experiment_code)
    meta_review_agent = MetaReviewAgent()
    supervisor = Supervisor()

    ideas = []

    # Idea generation and evaluation loop
    for i in range(args.iterations):
        print(f"Iteration {i+1}/{args.iterations}")

        # Generation
        new_ideas = generation_agent.run(ideas)  # Pass existing ideas for context
        ideas.extend(new_ideas)

        # Reflection
        for idea in ideas:
            reflection_agent.run(idea)

        # Ranking
        ranking_agent.run(ideas)

        # Proximity (for tournament organization in Ranking)
        proximity_agent.run(ideas)

        # Evolution
        evolved_ideas = []
        top_ideas = sorted(ideas, key=lambda x: x['ELO rating'], reverse=True)[:3] # evolve top 3
        for idea in top_ideas:
            evolved_ideas.extend(evolution_agent.run(idea, ideas)) # pass all ideas for context
        ideas.extend(evolved_ideas)

        # Meta-review
        meta_review_agent.run(ideas) # Apply meta review to all ideas

        # Supervisor - Simulate resource allocation based on meta-review, could involve adjusting agent prompts or focus
        supervisor.run(ideas, generation_agent, reflection_agent, ranking_agent, evolution_agent)

    # Sort ideas by ELO rating and save the top N
    top_ideas = sorted(ideas, key=lambda x: x['ELO rating'], reverse=True)[:args.num_ideas]
    save_ideas_to_json(top_ideas, ideas_file)

    print(f"Top {args.num_ideas} ideas saved to {ideas_file}")

if __name__ == "__main__":
    main()

                    self.update_elo(ideas[i], ideas[j], winner=True)

                    self.update_elo(ideas[i], ideas[j], winner=False)

    def update_elo(self, idea1, idea2, winner=True, k=32):
        rating1 = idea1["ELO rating"]
        rating2 = idea2["ELO rating"]

        # Calculate expected scores
        expected1 = 1 / (1 + 10 ** ((rating2 - rating1) / 400))
        expected2 = 1 / (1 + 10 ** ((rating1 - rating2) / 400))

        # Update ratings
        if winner:
            idea1["ELO rating"] = round(rating1 + k * (1 - expected1))
            idea2["ELO rating"] = round(rating2 + k * (0 - expected2))
        else:
            idea1["ELO rating"] = round(rating1 + k * (0 - expected1))
            idea2["ELO rating"] = round(rating2 + k * (1 - expected2))

**3. `utils.py`**

```python
import json
import os
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

def load_experiment_file(file_path):
    with open(file_path, "r") as f:
        return f.read()

def save_ideas_to_json(ideas, file_path):
    with open(file_path, "w") as f:
        json.dump(ideas, f, indent=4)

def calculate_similarity(text1, text2):
    # Calculate cosine similarity between two texts
    vectorizer = TfidfVectorizer()
    vectors = vectorizer.fit_transform([text1, text2])
    similarity = cosine_similarity(vectors[0], vectors[1])[0][0]
    return similarity
```
